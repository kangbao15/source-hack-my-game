                const shooter = p1;

                const dy = aimStart.y - aimCur.y;

                const drawAmt = Math.min(MAX_DRAW, dist);
                const powerPct = drawAmt / MAX_DRAW;
                const bow = shooter.bow;

                let simX = shooter.x;

                let simVy = Math.sin(angle) * speed;

                ctx.lineWidth = 3;
                ctx.setLineDash([4, 4]);
                ctx.beginPath();
                ctx.moveTo(simX, simY);

                for (let step = 0; step < 100; step++) {
                    simX += simVx * 0.06 * 16;
                    simY += simVy * 0.06 * 16;
                    simVy += GRAVITY * 16;
                    ctx.lineTo(simX, simY);
                    if (simY > GROUND_Y || simX > canvas.width || simX < 0) break;

                ctx.stroke();
                ctx.setLineDash([]);
                ctx.fillStyle = "#ff0055";
